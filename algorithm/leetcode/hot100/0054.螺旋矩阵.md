class Solution {
public:
    int dir[4][2]={{0,1},{1,0},{0,-1},{-1,0}};
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        int m=matrix.size();
        int n=matrix[0].size();
        int sz=m*n;
        vector<int>ans;
        int i=0,j=-1;
        for(int di=0;ans.size()<sz;di=(di+1)%4){
            for(int step=0;step<n;step++){
                i+=dir[di][0];
                j+=dir[di][1];
                ans.push_back(matrix[i][j]);
            }
            m--;
            swap(m,n);
        }
        return ans;
    }
};
