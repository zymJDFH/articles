知道前序和中序遍历 ，前序遍历的第一位为根，中间节点以该根为两边为左右子树。难度主要在于确认边界以及方案设计
```cpp
class Solution {
public:
    unordered_map<int,int>index;
    TreeNode *myBuild(vector<int>& preorder, vector<int>& inorder,int l1,int r1,int l2,int r2){
        if(l1>r1){
            return nullptr;
        }
        int root_idx=index[preorder[l1]];
        TreeNode *root=new TreeNode(preorder[l1]);
        int sz=root_idx-l2;
        root->left=myBuild(preorder,inorder,l1+1,l1+sz,l2,root_idx-1);
        root->right=myBuild(preorder,inorder,l1+sz+1,r1,root_idx+1,r2);
        return root;
    }
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        int n=preorder.size();
        for(int i=0;i<n;i++){
            index[inorder[i]]=i;
        }
        return myBuild(preorder,inorder,0,n-1,0,n-1);
    }
};
